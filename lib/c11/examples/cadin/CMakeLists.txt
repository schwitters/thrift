if(THRIFT_C11_COMPILER)
    set(c11_compiler "${THRIFT_C11_COMPILER}")
elseif(TARGET thrift-compiler AND NOT CMAKE_CROSSCOMPILING)
    set(c11_compiler "$<TARGET_FILE:thrift-compiler>")
else()
    message(FATAL_ERROR "Cadin requires THRIFT_C11_COMPILER pointing to a host compiler")
endif()
set(cadin_idls)
set(cadin_includes)
set(cadin_outputs)
set(cadin_commands)
set(cadin_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${cadin_generated}")
foreach(module common bom mdmp nodes nodes.index order rawrecord)
    set(directory "${THRIFT_C11_CADIN_ROOT}/de.cerpsw.cadin.integra.${module}/src/main/resources")
    list(APPEND cadin_includes -I "${directory}")
    if(module STREQUAL "rawrecord")
        set(stems RawRecord)
    elseif(module STREQUAL "nodes.index")
        set(stems CproIntegraNodesIndex)
    else()
        string(SUBSTRING "${module}" 0 1 first)
        string(TOUPPER "${first}" first)
        string(SUBSTRING "${module}" 1 -1 rest)
        set(stems "CproIntegra${first}${rest}")
        if(module STREQUAL "mdmp")
            list(APPEND stems CproIntegraMdmpBulk)
        endif()
    endif()
    foreach(stem IN LISTS stems)
        set(idl "${directory}/${stem}.thrift")
        if(NOT EXISTS "${idl}")
            message(FATAL_ERROR "Cadin IDL missing: ${idl}")
        endif()
        list(APPEND cadin_idls "${idl}")
        thrift_c11_file_stem(file_stem "${stem}")
        list(APPEND cadin_outputs "${cadin_generated}/${file_stem}_types.c" "${cadin_generated}/${file_stem}_types.h"
            "${cadin_generated}/${file_stem}_client.c" "${cadin_generated}/${file_stem}_client.h")
    endforeach()
endforeach()
foreach(idl IN LISTS cadin_idls)
    list(APPEND cadin_commands COMMAND "${c11_compiler}" ${cadin_includes}
        --gen c11 -out "${cadin_generated}" "${idl}")
endforeach()
add_custom_command(OUTPUT ${cadin_outputs} ${cadin_commands}
    DEPENDS "${c11_compiler}" ${cadin_idls} VERBATIM)
# Object library ensures the regression executable links every generated service.
add_library(thrift_c11_cadin_types OBJECT ${cadin_outputs})
target_include_directories(thrift_c11_cadin_types PUBLIC "${cadin_generated}")
target_link_libraries(thrift_c11_cadin_types PUBLIC thrift_c11)
set_target_properties(thrift_c11_cadin_types PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
thrift_c11_warnings(thrift_c11_cadin_types)
if(BUILD_TESTING)
    add_executable(thrift_c11_cadin_corpus ../../tests/cadin/test_cadin.c)
    target_link_libraries(thrift_c11_cadin_corpus PRIVATE thrift_c11_cadin_types)
    thrift_c11_warnings(thrift_c11_cadin_corpus)
    add_test(NAME thrift_c11_cadin_corpus COMMAND thrift_c11_cadin_corpus)
    set_tests_properties(thrift_c11_cadin_corpus PROPERTIES LABELS "c11_conformance;cadin" TIMEOUT 30)
endif()
if(THRIFT_C11_HTTP_CLIENT)
    add_executable(thrift_c11_cadin_client cadin_client.c)
    target_link_libraries(thrift_c11_cadin_client PRIVATE thrift_c11_cadin_types)
    set_target_properties(thrift_c11_cadin_client PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_definitions(thrift_c11_cadin_client PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    thrift_c11_warnings(thrift_c11_cadin_client)
    if(BUILD_TESTING)
        add_test(NAME thrift_c11_cadin_help COMMAND thrift_c11_cadin_client --help)
        find_package(Python3 QUIET COMPONENTS Interpreter)
        if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING AND
           EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../py/src/__init__.py")
            add_test(NAME thrift_c11_cadin_client COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/cadin/test_client.py"
                "$<TARGET_FILE:thrift_c11_cadin_client>")
            set_tests_properties(thrift_c11_cadin_client PROPERTIES LABELS "cadin;interop" TIMEOUT 30)
        endif()
    endif()
endif()
