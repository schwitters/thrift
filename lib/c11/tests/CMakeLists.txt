if(c11_compiler)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    add_subdirectory(conformance)
    if(Python3_Interpreter_FOUND)
        add_test(NAME thrift_c11_generator_errors COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/test_generator.py" "${c11_compiler}")
    endif()
    set(generated_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
    file(MAKE_DIRECTORY "${generated_dir}")
    add_custom_command(OUTPUT "${generated_dir}/fixture_types.c" "${generated_dir}/fixture_types.h"
        "${generated_dir}/common_types.c" "${generated_dir}/common_types.h"
        "${generated_dir}/fixture_server.h" "${generated_dir}/common_server.h"
        "${generated_dir}/fixture_client.c" "${generated_dir}/fixture_client.h"
        "${generated_dir}/common_client.c" "${generated_dir}/common_client.h"
        BYPRODUCTS "${generated_dir}/fixture_server.c" "${generated_dir}/common_server.c"
        COMMAND "${c11_compiler}" -r --gen c11:server_stubs -out "${generated_dir}" "${CMAKE_CURRENT_SOURCE_DIR}/fixture.thrift"
        DEPENDS "${c11_compiler}" fixture.thrift common.thrift VERBATIM)
    set(flat_outputs "${generated_dir}/flat_calls_types.c" "${generated_dir}/flat_calls_types.h"
        "${generated_dir}/flat_calls_client.c" "${generated_dir}/flat_calls_client.h")
    add_custom_command(OUTPUT ${flat_outputs}
        COMMAND "${c11_compiler}" --gen c11:flat_calls -out "${generated_dir}"
            "${CMAKE_CURRENT_SOURCE_DIR}/flat_calls.thrift"
        DEPENDS "${c11_compiler}" flat_calls.thrift VERBATIM)
    add_executable(thrift_c11_flat_calls_test test_flat_calls.c ${flat_outputs})
    target_include_directories(thrift_c11_flat_calls_test PRIVATE "${generated_dir}")
    target_link_libraries(thrift_c11_flat_calls_test PRIVATE thrift_c11)
    set_target_properties(thrift_c11_flat_calls_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    thrift_c11_warnings(thrift_c11_flat_calls_test)
    add_test(NAME thrift_c11_flat_calls COMMAND thrift_c11_flat_calls_test)
    add_library(thrift_c11_test_types STATIC "${generated_dir}/fixture_types.c" "${generated_dir}/common_types.c"
        "${generated_dir}/fixture_server.c" "${generated_dir}/common_server.c"
        "${generated_dir}/fixture_client.c" "${generated_dir}/common_client.c")
    target_include_directories(thrift_c11_test_types PUBLIC "${generated_dir}")
    target_link_libraries(thrift_c11_test_types PUBLIC thrift_c11)
    set_target_properties(thrift_c11_test_types PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    thrift_c11_warnings(thrift_c11_test_types)
    add_executable(thrift_c11_generated_test test_generated.c)
    target_link_libraries(thrift_c11_generated_test PRIVATE thrift_c11_test_types)
    set_target_properties(thrift_c11_generated_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    thrift_c11_warnings(thrift_c11_generated_test)
    add_test(NAME thrift_c11_generated COMMAND thrift_c11_generated_test)
    add_executable(thrift_c11_codec codec.c)
    target_link_libraries(thrift_c11_codec PRIVATE thrift_c11_test_types)
    set_target_properties(thrift_c11_codec PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    # This test deliberately uses ISO C fopen with checked results, not Annex K.
    if(MSVC)
        target_compile_definitions(thrift_c11_codec PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    thrift_c11_warnings(thrift_c11_codec)
    add_test(NAME thrift_c11_codec_help COMMAND thrift_c11_codec --help)
    if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING AND
       EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../py/src/__init__.py")
        add_test(NAME thrift_c11_python_interop COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/test_interop.py" "$<TARGET_FILE:thrift_c11_codec>")
        set_tests_properties(thrift_c11_python_interop PROPERTIES TIMEOUT 30)
    endif()
else()
    message(STATUS "C11 generator tests disabled: set THRIFT_C11_COMPILER to a host compiler")
endif()
